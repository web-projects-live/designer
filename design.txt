       constructor(type, x, y) {
           this.id = 'element_' + Date.now() + Math.floor(Math.random() * 1000);
           this.type = type;
           this.x = x;
           this.y = y;
           this.width = 100;
           this.height = 50;
           this.rotation = 0;
           this.selected = false;
           this.zIndex = 1;
       }
       
       draw(ctx) {
           // Base drawing functionality
           if (this.selected) {
               // Draw selection handles
               ctx.setLineDash([5, 5]);
               ctx.strokeStyle = '#0078d7';
               ctx.strokeRect(
                   this.x - 5, 
                   this.y - 5, 
                   this.width + 10, 
                   this.height + 10
               );
               ctx.setLineDash([]);
               
               // Draw resize handles
               this.drawResizeHandles(ctx);
           }
       }
       
       drawResizeHandles(ctx) {
           const handlePositions = [
               { x: this.x, y: this.y }, // top-left
               { x: this.x + this.width / 2, y: this.y }, // top-center
               { x: this.x + this.width, y: this.y }, // top-right
               { x: this.x + this.width, y: this.y + this.height / 2 }, // middle-right
               { x: this.x + this.width, y: this.y + this.height }, // bottom-right
               { x: this.x + this.width / 2, y: this.y + this.height }, // bottom-center
               { x: this.x, y: this.y + this.height }, // bottom-left
               { x: this.x, y: this.y + this.height / 2 } // middle-left
           ];
           
           ctx.fillStyle = '#ffffff';
           ctx.strokeStyle = '#0078d7';

               }

               // Extract shape type from tool name (e.g., shape_rectangle -> rectangle)
               const shapeType = this.activeTool.split('_')[1];
               
               // Create a new shape element
               const newShape = new ShapeElement(x, y, shapeType);
               this.elements.push(newShape);
               
               // Select the new element
               if (this.selectedElement) {
                   this.selectedElement.selected = false;
               }
               this.selectedElement = newShape;
               newShape.selected = true;
               
               // Update properties panel
               updatePropertiesPanel(newShape);
               
               this.render();
           }
       }
       
       handleMouseMove(e) {
           const rect = this.canvas.getBoundingClientRect();
           const x = e.clientX - rect.left;
           const y = e.clientY - rect.top;
           
           // Update cursor based on hover state
           this.updateCursor(x, y);
           
           if (!this.isDragging) return;
           
           const dx = x - this.dragStart.x;
           const dy = y - this.dragStart.y;
           
           if (this.resizeHandle) {
               // Resize the selected element
               this.selectedElement.resize(this.resizeHandle.position, dx, dy);
               this.dragStart = { x, y };
               this.render();
               updatePropertiesPanel(this.selectedElement);
           } else if (this.selectedElement) {
               // Move the selected element
               this.selectedElement.x += dx;
               this.selectedElement.y += dy;
               this.dragStart = { x, y };
               this.render();
               updatePropertiesPanel(this.selectedElement);
           }
       }
       
       handleMouseUp() {
           this.isDragging = false;
           this.resizeHandle = null;
       }
       
       updateCursor(x, y) {
           // Default cursor
           this.canvas.style.cursor = 'default';
           
           if (this.activeTool !== 'select') {
               this.canvas.style.cursor = 'crosshair';
               return;
           }
           
           // Check if hovering over a resize handle
           if (this.selectedElement) {
               const handle = this.selectedElement.getResizeHandleAt(x, y);
               if (handle) {
                   this.canvas.style.cursor = handle.cursor;
                   return;
               }
           }
           
           // Check if hovering over any element
           for (let i = this.elements.length - 1; i >= 0; i--) {
               if (this.elements[i].isPointInside(x, y)) {
                   this.canvas.style.cursor = 'move';
                   return;
               }
           }
       }
       
       setActiveTool(tool) {
           this.activeTool = tool;
       }

           // Draw grid (optional)
           this.drawGrid();

## Phase 2: User Interface Implementation (Estimated time: 2 days)

1. **Create Toolbar Manager (toolbar.js)**
   ```javascript
   function setupToolbar() {
       const toolbar = document.getElementById('toolbar');
       
       // Create tool buttons
       const tools = [
           { id: 'select', icon: '↖️', label: 'Select' },
           { id: 'text', icon: 'T', label: 'Text' },
           { id: 'shape_rectangle', icon: '⬜', label: 'Rectangle' },
           { id: 'shape_circle', icon: '⭕', label: 'Circle' },
           { id: 'shape_line', icon: '/', label: 'Line' }
       ];
       
       tools.forEach(tool => {
           const button = document.createElement('button');
           button.id = `tool-${tool.id}`;
           button.innerHTML = tool.icon;
           button.title = tool.label;
           button.addEventListener('click', () => {
               // Deactivate all tools
               document.querySelectorAll('#toolbar button').forEach(btn => {
                   btn.classList.remove('active');
               });
               
               // Activate selected tool
               button.classList.add('active');
               canvasManager.setActiveTool(tool.id);
           });
           
           toolbar.appendChild(button);
       });
       
       // Add export button
       const exportButton = document.createElement('button');
       exportButton.innerHTML = '💾';
       exportButton.title = 'Export Design';
       exportButton.style.marginLeft = 'auto';
       exportButton.addEventListener('click', () => {
           canvasManager.exportImage();
       });
       toolbar.appendChild(exportButton);
       
       // Activate select tool by default
       document.getElementById('tool-select').classList.add('active');
   }
   ```

2. **Create Properties Panel (propertiesPanel.js)**
   ```javascript
   function updatePropertiesPanel(element) {
       const panel = document.getElementById('propertiesPanel');
       panel.innerHTML = '';
       
       if (!element) {
           panel.innerHTML = 'No element selected';
           return;
       }
       
       const properties = element.getProperties();
       
       // Create header
       const header = document.createElement('h2');
       header.textContent = element.type.charAt(0).toUpperCase() + element.type.slice(1) + ' Properties';
       panel.appendChild(header);
       
       // Create property controls
       Object.entries(properties).forEach(([key, prop]) => {
           const group = document.createElement('div');
           group.className = 'property-row';
           
           const label = document.createElement('div');
           label.className = 'property-label';
           label.textContent = prop.label;
           
           const control = document.createElement('div');
           control.className = 'property-control';
           
           let input;
           
           switch (prop.type) {
               case 'number':
                   input = document.createElement('input');
                   input.type = 'number';
                   input.value = prop.value;
                   input.addEventListener('change', () => {
                       canvasManager.updateElementProperty(key, parseFloat(input.value));
                   });
                   break;
                   
               case 'text':
                   input = document.createElement('input');
                   input.type = 'text';
                   input.value = prop.value;
                   input.addEventListener('change', () => {
                       canvasManager.updateElementProperty(key, input.value);
                   });
                   break;
                   
               case 'color':
                   input = document.createElement('input');
                   input.type = 'color';
                   input.value = prop.value;
                   input.addEventListener('change', () => {
                       canvasManager.updateElementProperty(key, input.value);
                   });
                   break;
                   
               case 'checkbox':
                   input = document.createElement('input');
                   input.type = 'checkbox';
                   input.checked = prop.value;
                   input.addEventListener('change', () => {
                       canvasManager.updateElementProperty(key, input.checked);
                   });
                   break;
                   
               case 'select':
                   input = document.createElement('select');
                   prop.options.forEach(option => {
                       const optEl = document.createElement('option');
                       optEl.value = option;
                       optEl.textContent = option;
                       input.appendChild(optEl);
                   });
                   input.value = prop.value;
                   input.addEventListener('change', () => {
                       canvasManager.updateElementProperty(key, input.value);
                   });
                   break;
           }
           
           control.appendChild(input);
           group.appendChild(label);
           group.appendChild(control);
           panel.appendChild(group);
       });
       
       // Add delete button
       const deleteBtn = document.createElement('button');
       deleteBtn.textContent = 'Delete Element';
       deleteBtn.style.marginTop = '20px';
       deleteBtn.addEventListener('click', () => {
           canvasManager.elements = canvasManager.elements.filter(el => el !== element);
           canvasManager.selectedElement = null;
           updatePropertiesPanel(null);
           canvasManager.render();
       });
       panel.appendChild(deleteBtn);
   }
   ```
